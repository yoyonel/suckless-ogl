7def convert_doxygen_to_mkdocs(content: str, filename: str) -> str: 

8 """ 

9 Converts Doxygen Markdown syntax to MkDocs/CommonMark syntax. 

10 """ 

11 

12 # 1. Convert \code{.ext} to ```ext 

13 # Pattern: \code{.c} ... \endcode 

14 def replace_code_block(match): 

15 lang = match.group(1) if match.group(1) else "" 

16 code = match.group(2) 

17 # Remove leading dot if present in language extension (e.g. .c -> c) 

18 if lang.startswith('.'): 

19 lang = lang[1:] 

20 return f"```{lang}\n{code}```" 

21 

22 # Regex for \code{.ext} or \code 

23 # syntax: \code{.c} OR \code 

24 # We match \code(?:\{(\.\w+)\})?(.+?)\\endcode 

25 # FLAGS: DOTALL to match newlines 

26 content = re.sub(r'\\code(?:\{\s*(\.\w+)\s*\})?\n?(.+?)\\endcode', replace_code_block, content, flags=re.DOTALL) 

27 

28 # Handle simple \code without braces if the above didn't catch it all (or mixed usage) 

29 # The previous regex handles \code{.c} and also handles \code (group 1 is None) if we write it effectively. 

30 # But often Doxygen uses \code ... \endcode without extension. 

31 # Let's verify the first regex covers it. 

32 # The regex `\\code(?:\{\s*(\.\w+)\s*\})?` expects `{ .ext }` optionally. 

33 # If standard doxygen uses `\code` plain, it matches `(?:...)?` as empty. 

34 

35 # 2. Convert \dot ... \enddot to ```graphviz (or whatever the plugin expects) 

36 # Using 'graphviz' as language identifier for kroki or mkdocs-graphviz 

37 def replace_dot_block(match): 

38 code = match.group(1) 

39 return f"```graphviz\n{code}```" 

40 

41 content = re.sub(r'\\dot\n?(.+?)\\enddot', replace_dot_block, content, flags=re.DOTALL) 

42 

43 # 3. Convert \ref 

44 # Pattern: \ref class_name "Link Text" or just \ref class_name 

45 # Getting accurate links is hard without a full symbol table. 

46 # We will try to convert them to [Link Text](class_name) and hope relative linking works 

47 # or just keep them as text if elusive. 

48 # Best effort: \ref <ID> "<Text>" -> [<Text>](<ID>.md) 

49 # Note: Doxygen generates HTML, so file extensions are hidden. We assume we might map to .md files if they exist. 

50 

51 # Case 1: \ref ID "Text" 

52 content = re.sub(r'\\ref\s+(\w+)\s+"([^"]+)"', r'[\2](\1.md)', content) 

53 

54 # Case 2: \ref ID 

55 content = re.sub(r'\\ref\s+(\w+)', r'[\1](\1.md)', content) 

56 

57 # 4. Convert \note, \warning, etc. into Admonitions 

58 # Doxygen: \note text... 

59 # MkDocs: !!! note 

60 # text... 

61 # This is tricky because Doxygen commands might not end clearly. Usually they end at the paragraph end. 

62 # Simple strategy: replace `\note` with `!!! note\n ` and assume the user formats it manually or usage is simple. 

63 # However, Doxygen might use `\note` inline. 

64 # Let's just fix the block level ones which are common at start of lines. 

65 

66 def replace_admonition(match): 

67 type_ = match.group(1).lower() 

68 text = match.group(2) 

69 return f"!!! {type_}\n {text}" 

70 

71 # Regex: ^\s*\\(note|warning|attention)\s+(.*) 

72 # We apply this line by line? No, multi-line notes are harder. 

73 # Let's stick to simple replacements for now or manual fix. 

74 # Actually, Doxygen often uses \note followed by text using indentation. 

75 

76 # Let's leave \note as is? Or try a simple replace. 

77 # content = re.sub(r'\\(note|warning|attention|bug)', r'!!! \1', content) 

78 # Be careful with indentation. 

79 

80 # 5. Fix \page and \mainpage 

81 # \mainpage usually in README or distinct file. We might want to remove it. 

82 content = re.sub(r'\\mainpage.*', '', content) 

83 content = re.sub(r'\\page\s+\w+\s+.*', '', content) 

84 

85 # 6. Basic formatting 

86 # \b word -> **word** 

87 content = re.sub(r'\\b\s+(\w+)', r'**\1**', content) 

88 

89 # \c word -> `word` 

90 content = re.sub(r'\\c\s+(\w+)', r'`\1`', content) 

91 

92 return content 

93

97 print(f"Directory {root_dir} not found.") 

98 sys.exit(1)

101 print(f"Found {len(files)} markdown files to process.")
